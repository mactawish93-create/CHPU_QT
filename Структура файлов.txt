        │
        ├── mouse_events/        # --- Раздельные изолированные файлы событий мыши ---
        │   ├── __init__.py
        │   ├── click.py         # Обработка ЛКМ/СКМ, этапы кривой Безье и универсальный "взрыв" контуров ластиком
        │   ├── move.py          # Онлайн-трекинг мыши, резиновые нити фигур, тригонометрия дуг Безье
        │   ├── release.py       # Фиксация чистовой геометрии, запекание линий по Shift, вызов QTextEdit

        │
        ├── keyboard_system/     # --- Горячие клавиши, хоткеи и зажатые модификаторы ---
        │   ├── __init__.py

        │   ├── key_release.py   # Отпускание клавиш (выключение режимов магнита/орто)
