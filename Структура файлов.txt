CHPU_QT/
│
├── main.py                     # Главная точка входа. Инициализация QApplication и запуск ядра
├── config.py                   # Системный реестр вкладок (хедеров), пути, глобальные константы
├── styles.py                   # Единая таблица стилей QSS (Dark Mode, скругления кнопок, скроллбары)
│
├── core/                       # Системное инженерное ядро (общее для всех модулей, без UI)
│   ├── __init__.py
│   ├── gcode_generator.py      # Базовый класс и утилиты для форматирования G-кода (.tap)
│   └── machine_passport.py     # Хранение технологических параметров (безопасная Z, подачи F, обороты S)
│
├── ui/                         # Каркас главного интерфейса приложения
│   ├── __init__.py
│   ├── main_window.py          # Класс QMainWindow (держит только Хедер и Центральный контейнер вкладок)
│   └── header_widget.py        # Верхняя панель (логотип "Бани Бабочки" + динамические кнопки)
│
├── assets/                     # Статические ресурсы и медиа-файлы
│   ├── icons_3d/               # Современные 3D иконки
│   └── branding/               # Фирменные логотипы и графические элементы компании
│
└── modules/                    # ИЗОЛИРОВАННЫЕ РАБОЧИЕ МОДУЛИ (ВКЛАДКИ)
    │
    ├── tracking_paz/           # --- Модуль «Пазировка» ---
    │   ├── __init__.py
    │   ├── controller.py       # Диспетчер модуля (связывает UI параметров и логику холста)
    │   ├── model.py            # Математика расчета пазов и генерация G-кода
    │   ├── sidebar.py          # Левая панель параметров (внутри QScrollArea под 1366x768)
    │   └── canvas.py           # Правый 2D-холст визуализации (QGraphicsView)
    │   └── sub_views/          # Подмодули форм пазировки
    │       ├── __init__.py     
    │       ├── paz_banya_view.py       # Модуль пазировки Бани
    │       ├── paz_custom_view.py      # Модуль произвольной пазировки
    │       ├── paz_plane_view.py       # Модлуль выравнивания плоскости
    │
    ├── tracking_disks/         # --- Модуль «Диски» ---
    │   ├── __init__.py
    │   ├── controller.py       # Диспетчер раздела дисков
    │   ├── model.py            # Общая математика ламелей, гребенки, четверти и проема
    │   ├── sidebar.py          # Левая панель (выбор формы диска, глубин, смещений)
    │   ├── canvas.py           # Правый холст отрисовки геометрии досок
    │   ├── door_data.py        # Логика создания дверных проемов
    │   └── shapes/             # Подмодули конкретных форм (чистая математика без окон)
    │       ├── round.py        # Круглый диск
    │       ├── quadro.py       # Квадро диск
    │       ├── babochka.py     # Бабочка диск
    │       ├── viking.py       # Викинг диск
    │       └── quadrohouse.py  # КвадроХаус диск
    │
    ├── tracking_gcode/         # --- Модуль «Просмотр G-кода» (Бэкплоттер) ---
    │   ├── __init__.py
    │   ├── controller.py       # Логика загрузки файла и парсинга строк
    │   ├── sidebar.py          # Левая панель (Listbox со строками кода, кнопки, инфо)
    │   └── canvas.py           # Правый холст (парсинг траекторий G0/G1 и отрисовка векторов)
    │
    └── cad_editor/             # --- Модуль «Графический редактор» ---
        ├── __init__.py
        ├── controller.py       # Главный диспетчер (управляет состояниями, переключает инструменты)
        ├── toolbar.py          # Верхний горизонтальный тулбар редактора векторов
        ├── statusbar.py        # Нижняя панель состояния (Координаты X/Y, Зум %, Подсказки)
        ├── engine.py           # CAD-ядро (алгоритмы магнитной привязки, сетка 10мм)
        ├── canvas.py           # Огромное поле графики (QGraphicsView)
        │
        ├── mouse_events/       # Раздельные изолированные файлы под каждое событие мыши
        │   ├── __init__.py
        │   ├── click.py        # Обработка нажатий ЛКМ, ПКМ, СКМ (MouseButtonPress)
        │   ├── move.py         # Обработка движений мыши и отрисовка "резиновых нитей" (MouseMove)
        │   ├── release.py      # Фиксация геометрии при отпускании кнопок (MouseButtonRelease)
        │   └── wheel.py        # Масштабирование (Zoom) колесиком мыши (Wheel)
        │
        ├── keyboard_system/    # Горячие клавиши, хоткеи и зажатые модификаторы
        │   ├── __init__.py
        │   ├── key_press.py    # Нажатие клавиш (Ctrl для магнита, Shift для режима ОРТО)
        │   ├── key_release.py  # Отпускание клавиш (выключение режимов магнита/орто)
        │   └── shortcuts.py    # Глобальные хоткеи (Ctrl+Z отмена, Ctrl+C копирование, Esc сброс)
        │
        └── io_system/          # Независимая фабрика файлов (Импорт / Экспорт)
            ├── __init__.py
            ├── cam_project.py  # Чтение/Запись собственной файловой системы .cam на базе JSON
            ├── dxf_exchange.py # Импорт и экспорт векторов DXF AutoCAD
            ├── tap_export.py   # CAM-процессор генерации готового G-кода (.tap)
            ├── pdf_exchange.py # Экспорт чертежа в PDF для цеховой документации
            └── img_import.py   # Векторизатор картинок (IMG/PNG) для гравировки текста
